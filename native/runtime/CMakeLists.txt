cmake_minimum_required(VERSION 3.20)
project(glide_logits_fa_runner LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(LLAMA_CPP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../llama.cpp" CACHE PATH "Pinned official llama.cpp source")
set(GLIDE_LLAMA_REVISION "8e5a58df87ab59dfceb5b961ae26691e39f30c24")
set(GLIDE_EVIDENCE_CONTRACT "${CMAKE_CURRENT_SOURCE_DIR}/../EXECUTION_EVIDENCE_CONTRACT.json")
file(SHA256 "${GLIDE_EVIDENCE_CONTRACT}" GLIDE_EVIDENCE_CONTRACT_SHA256)
if(NOT GLIDE_EVIDENCE_CONTRACT_SHA256 STREQUAL "e9b7c3b70d9cd5af8ce776ffa8e955f1c23d9cef50c84b66a14fd72ae599a433")
  message(FATAL_ERROR "The immutable execution evidence contract does not match this implementation")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${GLIDE_EVIDENCE_CONTRACT}")
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_llama_source.py" "${LLAMA_CPP_DIR}"
  RESULT_VARIABLE GLIDE_SOURCE_STATUS OUTPUT_VARIABLE GLIDE_SOURCE_RESULT ERROR_VARIABLE GLIDE_SOURCE_ERROR)
if(NOT GLIDE_SOURCE_STATUS EQUAL 0)
  message(FATAL_ERROR "${GLIDE_SOURCE_ERROR}")
endif()
message(STATUS "${GLIDE_SOURCE_RESULT}")
add_custom_target(glide-verify-source
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_llama_source.py" "${LLAMA_CPP_DIR}"
  COMMENT "Verifying exact official llama.cpp source content before compilation" VERBATIM)
if(NOT EXISTS "${LLAMA_CPP_DIR}/include/llama.h")
  message(FATAL_ERROR "Obtain the pinned llama.cpp source; see README.md")
endif()
if(EXISTS "${LLAMA_CPP_DIR}/.git")
  execute_process(COMMAND git -C "${LLAMA_CPP_DIR}" rev-parse HEAD OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
elseif(EXISTS "${LLAMA_CPP_DIR}/PINNED_REVISION")
  file(STRINGS "${LLAMA_CPP_DIR}/PINNED_REVISION" LLAMA_REV LIMIT_COUNT 1)
else()
  message(FATAL_ERROR "Pinned source provenance missing; use fetch_llama.py or a clean pinned git checkout")
endif()
if(NOT LLAMA_REV STREQUAL GLIDE_LLAMA_REVISION)
  message(FATAL_ERROR "llama.cpp revision does not match the tested pin")
endif()

# CPU-only by default. CUDA support is opt-in at configure and run time.
# No automatic downloads or dynamic backend discovery.
option(GGML_CUDA "Build the optional CUDA backend (no GPU allocation at build time)" OFF)
if(GGML_CUDA)
  set(GGML_CUDA_FA ON CACHE BOOL "Required for the FA profile" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(OPT LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL GGML_METAL GGML_VULKAN GGML_SYCL GGML_OPENCL GGML_HIP GGML_RPC GGML_BACKEND_DL GGML_BLAS GGML_NATIVE)
  set(${OPT} OFF CACHE BOOL "" FORCE)
endforeach()
set(LLAMA_BUILD_COMMIT "${GLIDE_LLAMA_REVISION}" CACHE STRING "" FORCE)
add_subdirectory("${LLAMA_CPP_DIR}" "${CMAKE_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
foreach(TARGET llama ggml ggml-base ggml-cpu ggml-cuda)
  if(TARGET ${TARGET})
    add_dependencies(${TARGET} glide-verify-source)
  endif()
endforeach()

add_executable(glide-logits-fa glide_logits.cpp)
target_include_directories(glide-logits-fa PRIVATE "${LLAMA_CPP_DIR}/vendor")
target_link_libraries(glide-logits-fa PRIVATE llama)
if(GGML_CUDA)
  target_compile_definitions(glide-logits-fa PRIVATE GLIDE_CUDA_BUILD=1)
endif()
target_compile_definitions(glide-logits-fa PRIVATE GLIDE_EVIDENCE_CONTRACT_SHA256="${GLIDE_EVIDENCE_CONTRACT_SHA256}" GLIDE_LLAMA_REVISION="${GLIDE_LLAMA_REVISION}")
if(MSVC)
  target_compile_options(glide-logits-fa PRIVATE /W4)
else()
  target_compile_options(glide-logits-fa PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_executable(glide-protocol-tests tests/test_protocol.cpp)
target_include_directories(glide-protocol-tests PRIVATE "${LLAMA_CPP_DIR}/vendor" "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME protocol COMMAND glide-protocol-tests)
add_executable(glide-test-attention-control tests/attention_control.cpp)
target_include_directories(glide-test-attention-control PRIVATE "${LLAMA_CPP_DIR}/vendor" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(glide-test-attention-control PRIVATE llama)
add_test(NAME source-integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_source_verification.py")
if(Python3_FOUND)
  add_test(NAME cli-errors COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE:glide-logits-fa>")

  if(NUMPY_STATUS EQUAL 0)
    add_test(NAME tiny-model-cpu COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tiny_model.py" "$<TARGET_FILE:glide-logits-fa>")
    set_tests_properties(tiny-model-cpu PROPERTIES ENVIRONMENT "LLAMA_CPP_DIR=${LLAMA_CPP_DIR}" TIMEOUT 120)
    add_test(NAME grouped-noncausal-cpu COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_grouped_noncausal.py" "$<TARGET_FILE:glide-logits-fa>" "$<TARGET_FILE:glide-test-attention-control>")
    set_tests_properties(grouped-noncausal-cpu PROPERTIES ENVIRONMENT "LLAMA_CPP_DIR=${LLAMA_CPP_DIR}" TIMEOUT 120)
  endif()
endif()

if(NUMPY_STATUS EQUAL 0)
  add_test(NAME production-shape-fa-cpu COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fa_production_control.py"
    --runner "$<TARGET_FILE:glide-logits-fa>" --control "$<TARGET_FILE:glide-test-attention-control>" --backend cpu
    --output-dir "${CMAKE_BINARY_DIR}/production-shape-control")
  set_tests_properties(production-shape-fa-cpu PROPERTIES ENVIRONMENT "LLAMA_CPP_DIR=${LLAMA_CPP_DIR}" TIMEOUT 600)
endif()

target_link_libraries(glide-execution-evidence-tests PRIVATE llama)
add_test(NAME execution-evidence-admission COMMAND glide-execution-evidence-tests)
